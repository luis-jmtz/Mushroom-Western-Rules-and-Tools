def parse_traits(traits_list,needNameList):
	output_text = ""

	names_list = {}
      
	id_list = []

	for index, row in traits.iterrows():
		if traits['id'].iloc[index] in traits_list:
			trait_name = traits['name'].iloc[index]
			trait_descript = traits['description'].iloc[index]
			output_text += f"{trait_name}: {trait_descript} \n\n"
            
			trait_id = int(traits.loc[index, "id"])

			if needNameList:
				names_list[trait_name] = trait_id
        	
			else:
				id_list.append(trait_id)

	if needNameList:
		return output_text, names_list
	
	return output_text, id_list

def update_skill_points(number):
    st.session_state.skill_points -= number

def update_skills(skill_name, points):
    skills.loc[skills["name"] == skill_name, "lvl"] = points
    

def gen_skills(rows, attribute):
    for index, row in rows.iterrows():

        if row["attribute"] == attribute:

            name = row["name"]

            expertise = row["expertise"]
            if expertise == 1:
                max_value = 2
            else:
                max_value = 1

            base_points = int(skills.loc[skills["name"] == name, "lvl"]) 
            # used if an ability/trait grants the user extra skill points

            points = st.number_input(f"{name}", key=name,step=1,min_value=base_points,max_value=max_value)

            update_skills(name,points)
            update_skill_points(points)
